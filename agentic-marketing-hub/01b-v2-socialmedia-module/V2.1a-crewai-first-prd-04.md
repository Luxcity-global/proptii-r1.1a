// Frontend CrewAI Service
class CrewAIService {
  async generatePropertyAsset(
    propertyData: PropertyData
  ): Promise<AssetResult> {
    const crew = new PropertyMarketingCrew();
    return await crew.process(propertyData);
  }

  async optimizeContent(
    content: string,
    platform: Platform
  ): Promise<OptimizedContent> {
    return await this.contentOptimizer.optimize(content, platform);
  }

  async validateCompliance(
    content: MarketingContent
  ): Promise<ComplianceReport> {
    return await this.complianceAgent.validate(content);
  }
}
